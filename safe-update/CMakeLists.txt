cmake_minimum_required(VERSION 3.20)
project(SafeUpdate VERSION 0.1.0 LANGUAGES CXX)
if(NOT UNIX)
  message(FATAL_ERROR "The safe-update reference currently requires POSIX filesystem APIs")
endif()
include(CTest)
include(GNUInstallDirs)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
if(OPENSSL_VERSION VERSION_GREATER_EQUAL 4)
  message(FATAL_ERROR "OpenSSL 4 has not been qualified")
endif()
add_library(safe_update src/update.cpp)
target_compile_features(safe_update PUBLIC cxx_std_20)
target_include_directories(safe_update PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(safe_update PUBLIC OpenSSL::Crypto)
target_compile_options(safe_update PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
option(SAFE_UPDATE_SANITIZERS "Run with address/undefined sanitizers" OFF)
if(SAFE_UPDATE_SANITIZERS)
 target_compile_options(safe_update PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(safe_update PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
 add_executable(safe_update_tests tests/update.cpp)
 target_link_libraries(safe_update_tests PRIVATE safe_update)
 target_compile_options(safe_update_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
 add_test(NAME safe_update COMMAND safe_update_tests)
endif()
install(TARGETS safe_update EXPORT SafeUpdateTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SafeUpdateTargets NAMESPACE SafeUpdate:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SafeUpdate)
include(CMakePackageConfigHelpers)
configure_package_config_file(SafeUpdateConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SafeUpdateConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SafeUpdate)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SafeUpdateConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SafeUpdate)
