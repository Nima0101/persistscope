cmake_minimum_required(VERSION 3.20)
project(PersistScope VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
option(PERSISTScope_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(persistscope_core src/persistscope.cpp)
add_library(PersistScope::core ALIAS persistscope_core)
set_target_properties(persistscope_core PROPERTIES EXPORT_NAME core)
target_compile_features(persistscope_core PUBLIC cxx_std_20)
target_include_directories(persistscope_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
function(persistscope_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
  endif()
  if(PERSISTScope_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "Use Clang or GCC for ASan/UBSan")
    endif()
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()
persistscope_warnings(persistscope_core)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(core_tests tests/core.cpp)
  target_link_libraries(core_tests PRIVATE persistscope_core Threads::Threads)
  persistscope_warnings(core_tests)
  add_test(NAME core COMMAND core_tests)
endif()
add_executable(persistscope src/cli.cpp)
target_link_libraries(persistscope PRIVATE persistscope_core)
set_target_properties(persistscope PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
persistscope_warnings(persistscope)
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:persistscope>)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/PersistScopeConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/PersistScopeConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PersistScope)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/PersistScopeConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(TARGETS persistscope_core persistscope EXPORT PersistScopeTargets
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PersistScopeTargets NAMESPACE PersistScope:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PersistScope)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PersistScopeConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/PersistScopeConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PersistScope)
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/persistscope)
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/persistscope/examples)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "PersistScope maintainers")
set(CPACK_GENERATOR ZIP)
include(CPack)
